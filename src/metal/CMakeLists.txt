# momoe-metal: Momoe's engine for Apple Silicon, built with Momoe's copy of ggml (patched: patch_ggml_pause.py).
# On Linux the same engine builds against ggml's CUDA backend as momoe-cuda (MOMOE_CUDA): no pause points there, the
# scheduler's callback stops after each layer's router instead.
cmake_minimum_required(VERSION 3.21)
option(MOMOE_EXPERIMENTAL_SM52 "Experimental Maxwell SM52 build with CUDA 12" OFF)
if(APPLE)
  if(MOMOE_EXPERIMENTAL_SM52)
    message(FATAL_ERROR "MOMOE_EXPERIMENTAL_SM52 requires Linux and CUDA 12")
  endif()
  project(momoe_metal C CXX OBJC)
  set(GGML_METAL ON CACHE BOOL "" FORCE)
  set(GGML_METAL_EMBED_LIBRARY ON CACHE BOOL "" FORCE)
else()
  project(momoe_cuda C CXX)
  if(MOMOE_EXPERIMENTAL_SM52)
    find_package(CUDAToolkit REQUIRED)
    if(CUDAToolkit_VERSION VERSION_LESS "12.4" OR CUDAToolkit_VERSION VERSION_GREATER_EQUAL "13.0")
      message(FATAL_ERROR "Maxwell SM52 requires CUDA 12.4 or newer within CUDA 12; CUDA 13 removed Maxwell compilation")
    endif()
    set(CMAKE_CUDA_ARCHITECTURES 52 CACHE STRING "Maxwell CUDA architecture" FORCE)
    message(WARNING "Experimental Maxwell SM52: run momoe-cuda-sm52-check on the target GPU before serving")
  endif()
  set(GGML_CUDA ON CACHE BOOL "" FORCE)
  set(GGML_CUDA_GRAPHS ON CACHE BOOL "" FORCE)           # a reused decode graph is replayed (ggml: off unless asked)
endif()
set(CMAKE_CXX_STANDARD 17)
set(GGML_BLAS OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
add_subdirectory(ggml)
if(APPLE)
  add_executable(momoe-metal momoe_metal.cpp)
  target_link_libraries(momoe-metal PRIVATE ggml)
else()
  add_executable(momoe-cuda momoe_metal.cpp)
  find_package(CUDAToolkit REQUIRED)
  target_compile_definitions(momoe-cuda PRIVATE MOMOE_CUDA=1)
  target_link_libraries(momoe-cuda PRIVATE ggml CUDA::cudart)
  if(MOMOE_EXPERIMENTAL_SM52)
    add_executable(momoe-cuda-sm52-check Test-CudaMatrix.cpp)
    target_link_libraries(momoe-cuda-sm52-check PRIVATE ggml CUDA::cudart)
  endif()
endif()
